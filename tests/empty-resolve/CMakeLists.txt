cmake_minimum_required(VERSION 3.25)
project(overg_resolve_check LANGUAGES CXX)

find_package(rexglue 0.10.1 EXACT CONFIG REQUIRED)
set(SDK_BUILD "" CACHE PATH "Patched ReXGlue build from build-resolve-plugin.ps1")
set(plugin_objects "${SDK_BUILD}/src/graphics/CMakeFiles/rexgpu-xenos.dir/RelWithDebInfo")
file(GLOB_RECURSE objects "${plugin_objects}/*.cpp.obj")
if(NOT EXISTS "${plugin_objects}/util/draw.cpp.obj")
    message(FATAL_ERROR "Build the patched RelWithDebInfo GPU plugin first.")
endif()
add_executable(overg-resolve-check main.cpp ${objects})
target_compile_features(overg-resolve-check PRIVATE cxx_std_23)
target_link_libraries(overg-resolve-check PRIVATE rex::runtime rex::xxhash d3d12 dxgi dxguid)
rexglue_apply_target_settings(overg-resolve-check)
